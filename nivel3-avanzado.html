Bucles 'for' y 'while' 🔄

Automatiza tareas repetitivas

🔒 Bloqueada

💡 Bucles

# Bucle FOR - repite un número fijo de veces
for i in range(5):
    print(f"Contando: {i}")

# Bucle WHILE - repite mientras sea verdadero
contador = 0
while contador < 3:
    print(f"Número: {contador}")
    contador += 1

# Recorrer listas
frutas = ["manzana", "banana", "naranja"]
for fruta in frutas:
    print(fruta)
